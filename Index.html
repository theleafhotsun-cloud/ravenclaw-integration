  <div class="login-box">
    <h2>Ravenclaw Integration System</h2>
    <input type="text" id="username" placeholder="Enter your username">
    <!-- Login Button -->
    <button id="btn-login" onclick="processLogin()">Login</button>
    <div id="message"></div>
  </div>

  <script>
    // INSERT YOUR APPS SCRIPT URL HERE
    const API_URL = "https://script.google.com/macros/s/AKfycbzaVAkUg-1choWXhQ93tu3HawpBk1iD7dWh6TdGMRJ6Okt4kxOsHSyqAd7lUcspm40g/exec";

    function processLogin() {
      const usernameInput = document.getElementById('username').value;
      const messageDiv = document.getElementById('message');
      const btnLogin = document.getElementById('btn-login');

      if(!usernameInput) {
        messageDiv.innerHTML = "Username cannot be empty!";
        messageDiv.style.color = "red";
        return;
      }

      // Change button state during loading
      btnLogin.innerHTML = "Authenticating...";
      btnLogin.disabled = true;
      messageDiv.innerHTML = "";

      // Send data to Apps Script
      fetch(API_URL, {
        method: "POST",
        // Text/plain is required for Apps Script to avoid CORS preflight issues
        headers: { "Content-Type": "text/plain;charset=utf-8" },
        body: JSON.stringify({ username: usernameInput })
      })
      .then(response => response.json())
      .then(result => {
        if (result.status === "success") {
          messageDiv.innerHTML = "Login Successful! Redirecting...";
          messageDiv.style.color = "green";
         
          // SAVE SESSION DATA IN BROWSER
          localStorage.setItem("session_username", result.data.username);
          localStorage.setItem("session_role", result.data.role);

          // REDIRECT TO DASHBOARD BASED ON ROLE
          setTimeout(() => {
            if (result.data.role === "Headhouse") {
              window.location.href = "headhouse-dashboard.html";
            } else if (result.data.role === "Prefect") {
              window.location.href = "prefect-dashboard.html";
            } else {
              window.location.href = "general-dashboard.html"; // Default fallback
            }
          }, 1000);

        } else {
          messageDiv.innerHTML = result.message;
          messageDiv.style.color = "red";
          btnLogin.innerHTML = "Login";
          btnLogin.disabled = false;
        }
      })
      .catch(error => {
        messageDiv.innerHTML = "Failed to connect to the server.";
        messageDiv.style.color = "red";
        btnLogin.innerHTML = "Login";
        btnLogin.disabled = false;
      });
    }
  </script>
